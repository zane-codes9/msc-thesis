import ee
from datetime import datetime, timedelta

# Toronto bbox in lat/lon (padded ±0.05° for edge polygons)
toronto_bbox = ee.Geometry.Rectangle([-79.69, 43.53, -79.07, 44.10])

# Daymet V4 in Earth Engine
daymet = ee.ImageCollection('NASA/ORNL/DAYMET_V4')

# Filter: 2015-2019, May-Sep, tmax + tmin
def filter_warm_season(year):
    start = ee.Date(f'{year}-05-01')
    end   = ee.Date(f'{year}-09-30').advance(1, 'day')  # inclusive of Sep 30
    return (daymet
            .filterDate(start, end)
            .filterBounds(toronto_bbox)
            .select(['tmax', 'tmin']))

# Collect all years
all_years = ee.ImageCollection([])
for yr in range(2015, 2020):
    yearly = filter_warm_season(yr)
    all_years = all_years.merge(yearly)

n_images = all_years.size().getInfo()
print(f"Daymet images queued: {n_images}")  # expect 765 (153 days × 5 years)

import ee
import pandas as pd

# Load the DA asset
da_fc = ee.FeatureCollection('projects/gen-lang-client-0569405164/assets/toronto_da_shp')
print(f"DA polygons loaded: {da_fc.size().getInfo()}")  # expect 7716

# Daymet image collection (already filtered earlier; re-build here for clarity)
daymet = (ee.ImageCollection('NASA/ORNL/DAYMET_V4')
          .filterDate('2015-05-01', '2019-10-01')
          .filterBounds(da_fc.geometry().bounds())
          .select(['tmax', 'tmin']))

# Filter to warm-season months only (May-Sep)
def keep_warm(img):
    m = ee.Date(img.get('system:time_start')).get('month')
    cond = m.gte(5).And(m.lte(9))
    return img.set('keep', cond)

daymet_warm = daymet.map(keep_warm).filter(ee.Filter.eq('keep', 1))
n_days = daymet_warm.size().getInfo()
print(f"Warm-season Daymet days: {n_days}")  # expect 765

import ee

# For each Daymet day, reduce tmax/tmin over each DA polygon
def reduce_one_day(img):
    date_str = ee.Date(img.get('system:time_start')).format('YYYY-MM-dd')

    reduced = img.reduceRegions(
        collection=da_fc,
        reducer=ee.Reducer.mean(),
        scale=1000,  # Daymet native resolution
        tileScale=4  # split into smaller tiles to avoid memory errors
    )

    # Attach the date as a property on every feature
    return reduced.map(lambda f: f.set('date', date_str))

# Map over all days, flatten into one big FeatureCollection
all_reductions = daymet_warm.map(reduce_one_day).flatten()

# Configure export
task = ee.batch.Export.table.toDrive(
    collection=all_reductions,
    description='toronto_daymet_2015_2019',
    folder='thesis/dlnm-pilot',          # subfolder inside MyDrive
    fileNamePrefix='toronto_daymet_2015_2019',
    fileFormat='CSV',
    selectors=['DAUID', 'date', 'tmax', 'tmin']  # only keep these columns
)

task.start()
print(f"Task started. ID: {task.id}")
print("Monitor at: https://code.earthengine.google.com/tasks")

    print(f"  {task.status()['state']}... {time.strftime('%H:%M:%S')}")
    time.sleep(30)
print(f"Final state: {task.status()['state']}")

"""nice. now daymet reduce, MTL + VAN. boxes now derived from the actual DA polygons (verified, R st_bbox).

import ee

city_boxes = {
    "montreal_da":  ee.Geometry.Rectangle([-74.38, 45.17, -73.08, 46.02]),
    "vancouver_da": ee.Geometry.Rectangle([-123.48, 48.95, -122.36, 49.62]),
}
city_prefix = {"montreal_da": "montreal", "vancouver_da": "vancouver"}

tasks = {}
for city, bbox in city_boxes.items():
    da_fc = ee.FeatureCollection(f'projects/gen-lang-client-0569405164/assets/{city}')

    daymet = (ee.ImageCollection('NASA/ORNL/DAYMET_V4')
              .filterDate('2015-05-01', '2019-10-01')
              .filterBounds(bbox)
              .select(['tmax', 'tmin']))

    def keep_warm(img):
        m = ee.Date(img.get('system:time_start')).get('month')
        return img.set('keep', m.gte(5).And(m.lte(9)))
    daymet_warm = daymet.map(keep_warm).filter(ee.Filter.eq('keep', 1))

    def reduce_one_day(img):
        date_str = ee.Date(img.get('system:time_start')).format('YYYY-MM-dd')
        reduced = img.reduceRegions(collection=da_fc,
                                    reducer=ee.Reducer.mean(),
                                    scale=1000, tileScale=4)
        return reduced.map(lambda f: f.set('date', date_str))
    all_reductions = daymet_warm.map(reduce_one_day).flatten()

    prefix = city_prefix[city]
    task = ee.batch.Export.table.toDrive(
        collection=all_reductions,
        description=f'{prefix}_daymet_2015_2019',
        folder='thesis/dlnm-pilot',
        fileNamePrefix=f'{prefix}_daymet_2015_2019',
        fileFormat='CSV',
        selectors=['DAUID', 'date', 'tmax', 'tmin'])
    task.start()
    tasks[city] = task
    print(f"{city}: task started — {task.id}")
