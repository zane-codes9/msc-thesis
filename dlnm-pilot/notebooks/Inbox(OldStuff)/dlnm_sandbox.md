"""### BLOCK 2: Simulator"""

n_cma <- 20
n_da <- 50
n_years <- 10
start_date <- as.Date("2014-01-01")
end_date <- as.Date("2023-12-31")
dates <- seq (start_date, end_date, by = "day")
n_days <- length(dates)

cat("Sim grid", n_cma, "CMA x", n_da, "DAs x", n_days, "days, =",
  formatC(n_cma * n_da * n_days, big.mark = ","), "rows\n")

cma_mean_temp <- rnorm(n_cma, mean = 15, sd = 3)
cma_mean_vuln <- rnorm(n_cma, mean = 0, sd = 0.5)

# DA-level vulnerabillity (DA-only cell of the 2x2)
# Each DA gets a score drawn around its CMA mean. Within-CMA spread is what downscaling exists to leverage
da_id <- 1:(n_cma * n_da)
da_cma <- rep(1:n_cma, each = n_da)
da_vuln <- cma_mean_vuln[da_cma] + rnorm(n_cma * n_da, 0, 1)

# Da-level baseline population (drives baseline deaths)
da_pop <- round(runif(n_cma * n_da, 300, 800))

# Temperature series per DA
# Seasonal sinusoid + AR(1) noise. Each DA has temperature varying around its CMA mean, with seasonal cycle
day_of_year <- as.numeric(format(dates, "%j"))
seasonal <- -12 * cos(2 * pi * day_of_year / 365) # peaks in summer

# Build temperature for every DA-day. This produces a giant matrix.

temp_matrix <- matrix(NA, nrow = n_cma * n_da, ncol = n_days)
for (i in 1: (n_cma * n_da)) {
  cma_temp <- cma_mean_temp[da_cma[i]]
  # AR(1) noise so consecutive days are correlated
  noise <- as.numeric(arima.sim(list(ar = 0.7), n = n_days, sd = 2))
  temp_matrix[i, ] <- cma_temp + seasonal + noise
}

cat("Temperature range across sim:",
    round(min(temp_matrix), 1), "to", round(max(temp_matrix), 1), "°C\n")

# --- The true temp-resp (this is what we are trying to recover)
# Baseline: J-curve that is gentle. Cold has a small effect, heat is the signal. Vulnerability adds a hot-temp kick proportional to V

beta_true <- function(temp, vuln) {
  base <- 0.002 * (temp - 18)^2 * (temp > 24) # quadratic hot effect above 24°C
  vuln_kick <- 0.04 * pmax(temp - 25, 0) * vuln # extra effect from vuln above 25°C
  base + vuln_kick
}


# --- Generate death counts ---
# log(λ) = log(baseline_rate x pop) + β_true(T, V)
# Y ~ Poisson(λ)

cat("Generating deaths (this takes a few seconds)...\n")
baseline_daily_rate <- 8 / 1000 / 365 # 8 deaths per 1000 per year

deaths_matrix <- matrix(NA, nrow = n_cma * n_da, ncol = n_days)
for (i in 1: (n_cma * n_da)) {
  log_baseline <- log(baseline_daily_rate * da_pop[i])
  log_lambda <- log_baseline + beta_true(temp_matrix[i, ], da_vuln[i])
  deaths_matrix[i, ] <- rpois(n_days, exp(log_lambda))
}
