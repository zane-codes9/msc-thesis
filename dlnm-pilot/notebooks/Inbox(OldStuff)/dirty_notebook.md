# Section 3 — Daymet daily temperature for Toronto, May–Sep 2015–2019
# Pull single tile = native Daymet 1km grid for Toronto bbox
# Cached per year. If a year fails, only that year reruns.

dir.create("/content/daymet", showWarnings = FALSE)

# Toronto CMA bbox (NAD83 lat/lon). Padded ±0.05° for edge polygons.
# Daymet uses lon/lat (WGS84) — close enough to NAD83 for raster sampling.
toronto_bbox <- c(
  lon_min = -79.69,
  lat_min =  43.53,
  lon_max = -79.07,
  lat_max =  44.10
)

years <- year_start:year_end

for (yr in years) {
  out_nc <- sprintf("/content/daymet/tmax_%d.nc", yr)
  if (file.exists(out_nc)) {
    cat(sprintf("Year %d: cached, skipping\n", yr))
    next
  }
  cat(sprintf("Year %d: downloading...\n", yr))
  daymetr::download_daymet_ncss(
    location = c(toronto_bbox["lat_max"], toronto_bbox["lon_min"],
                 toronto_bbox["lat_min"], toronto_bbox["lon_max"]),
    start    = yr,
    end      = yr,
    param    = "tmax",
    path     = "/content/daymet",
    silent   = FALSE
  )
  # daymetr names files like "tmax_daily_2015_ncss.nc" — rename for consistency
  src <- sprintf("/content/daymet/tmax_daily_%d_ncss.nc", yr)
  if (file.exists(src)) file.rename(src, out_nc)
}

# Also pull tmin (we'll average to tmean later)
for (yr in years) {
  out_nc <- sprintf("/content/daymet/tmin_%d.nc", yr)
  if (file.exists(out_nc)) {
    cat(sprintf("Year %d tmin: cached, skipping\n", yr))
    next
  }
  cat(sprintf("Year %d tmin: downloading...\n", yr))
  daymetr::download_daymet_ncss(
    location = c(toronto_bbox["lat_max"], toronto_bbox["lon_min"],
                 toronto_bbox["lat_min"], toronto_bbox["lon_max"]),
    start    = yr,
    end      = yr,
    param    = "tmin",
    path     = "/content/daymet",
    silent   = FALSE
  )
  src <- sprintf("/content/daymet/tmin_daily_%d_ncss.nc", yr)
  if (file.exists(src)) file.rename(src, out_nc)
}

# Direct download from ORNL THREDDS, bypassing daymetr entirely
dir.create("/content/daymet", showWarnings = FALSE)

for (yr in years) {
  for (param in c("tmax", "tmin")) {
    out_nc <- sprintf("/content/daymet/%s_%d.nc", param, yr)
    if (file.exists(out_nc) && file.size(out_nc) > 1e6) {
      cat(sprintf("%s %d: cached, size %.1f MB\n",
                  param, yr, file.size(out_nc)/1e6))
      next
    }
    url <- sprintf(
      "https://thredds.daac.ornl.gov/thredds/ncss/ornldaac/1840/daymet_v4_daily_na_%s_%d.nc?var=lat&var=lon&var=%s&north=44.10&west=-79.69&east=-79.07&south=43.53&disableProjSubset=on&horizStride=1&time_start=%d-05-01T12:00:00Z&time_end=%d-09-30T12:00:00Z&timeStride=1&accept=netcdf",
      param, yr, param, yr, yr
    )
    cat(sprintf("Downloading %s %d ...\n", param, yr))
    download.file(url, out_nc, mode = "wb", quiet = FALSE)
    cat(sprintf("  -> %.1f MB\n", file.size(out_nc)/1e6))
  }
}
