# === Cell 7e: Chunked DA pull, looping over CSDs in Toronto CMA ===
# Strategy: pull each CSD's DAs (with the 18 age vectors) one at a time.
# cancensus caches each call, so re-runs are free.
# If a single CSD fails on quota, we'll chunk that one further.

csd_codes <- toronto_csds_dt[order(-pop)]$region   # largest first, so failures surface early

results <- list()
fail_log <- character(0)

for (i in seq_along(csd_codes)) {
  csd <- csd_codes[i]
  csd_name <- toronto_csds_dt[region == csd, name]
  csd_pop  <- toronto_csds_dt[region == csd, pop]
  cat(sprintf("[%2d/%2d] %s (CSD %s, pop %s)... ",
              i, length(csd_codes), csd_name, csd, format(csd_pop, big.mark=",")))

  t0 <- Sys.time()
  res <- tryCatch({
    cancensus::get_census(
      dataset    = census_year,
      regions    = list(CSD = csd),
      vectors    = all_vecs,
      level      = "DA",
      geo_format = "sf",
      quiet      = TRUE
    )
  }, error = function(e) {
    cat("FAILED: ", conditionMessage(e), "\n")
    fail_log <<- c(fail_log, csd)
    return(NULL)
  })

  if (!is.null(res)) {
    t1 <- Sys.time()
    cat(sprintf("✓ %d DAs in %.1fs\n", nrow(res), as.numeric(t1 - t0, units = "secs")))
    results[[csd]] <- res
  }
}

# =====================================================================
# Cell 43 — Daymet daily temperature for Toronto, warm season 2015–2019
# =====================================================================
# Downloads tmax + tmin from NASA ORNL DAAC via daymetr::download_daymet_ncss()
# Computes tmean = (tmax + tmin) / 2 in-memory before saving.
# One netCDF per year per variable, cached to /content/daymet/.
# Resumes on failure: skip download if file exists.
# =====================================================================

dir.create("/content/daymet", showWarnings = FALSE)

# --- bbox from Toronto polygons, padded ±0.05° for edge DAs ---
toronto_bbox <- sf::st_bbox(sf::st_transform(toronto_da_full, 4326))
pad <- 0.05
bbox_pad <- c(
  xmin = unname(toronto_bbox["xmin"]) - pad,
  ymin = unname(toronto_bbox["ymin"]) - pad,
  xmax = unname(toronto_bbox["xmax"]) + pad,
  ymax = unname(toronto_bbox["ymax"]) + pad
)
cat("BBox (lon/lat, padded):\n"); print(round(bbox_pad, 3))

# daymetr expects c(top, left, bottom, right) = c(ymax, xmin, ymin, xmax)
loc_box <- c(bbox_pad["ymax"], bbox_pad["xmin"],
             bbox_pad["ymin"], bbox_pad["xmax"])

# --- download tmax and tmin, year by year ---
vars_to_pull <- c("tmax", "tmin")
years        <- year_start:year_end

for (yr in years) {
  for (v in vars_to_pull) {
    fpath <- sprintf("/content/daymet/%s_%d_ncss.nc", v, yr)
    if (file.exists(fpath)) {
      cat(sprintf("  [skip] %s already exists\n", basename(fpath)))
      next
    }
    cat(sprintf("  [pull] %s %d ... ", v, yr))
    t0 <- Sys.time()
    daymetr::download_daymet_ncss(
      location  = loc_box,
      start     = yr,
      end       = yr,
      param     = v,
      frequency = "daily",
      path      = "/content/daymet",
      silent    = TRUE
    )
    # daymetr writes as <param>_daily_<yr>_ncss.nc; rename for consistency
    written <- list.files("/content/daymet",
                          pattern = sprintf("%s_daily_%d", v, yr),
                          full.names = TRUE)
    if (length(written) == 1 && written != fpath) {
      file.rename(written, fpath)
    }
    cat(sprintf("done (%.1fs)\n", as.numeric(Sys.time() - t0, units = "secs")))
  }
}
