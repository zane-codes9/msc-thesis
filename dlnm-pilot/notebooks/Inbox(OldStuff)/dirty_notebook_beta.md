# === 01_simulate.R (cont'd) — Temperature-modulated mortality ===
# Function builds DA-day expected counts given a temperature vector

# We need DA-specific MMT. Without temp data yet, we anchor at a placeholder
# (Toronto's expected ~18C July-mean) and re-anchor at DA-level later.
# Real DGP: log_RR(T) = base_curve(T) × (1 + 0.4*F1*heat_ind(T) + 0.3*F2*cold_ind(T) + 0.2*F3)

# --- True U-shaped base curve (per Gasparrini 2022 spirit) ---
# MMT anchored at 80th percentile of CMA temp distribution
base_log_rr <- function(temp, mmt) {
  # quadratic U above and below MMT, asymmetric (heat steeper than cold for warm-season pilot)
  ifelse(temp > mmt,
         0.03 * (temp - mmt)^2,   # heat slope
         0.01 * (temp - mmt)^2)   # cold slope (less relevant warm-season-only)
}

# --- Lag distribution: peak at lag 0-3 for heat ---
# Distribute the daily effect across 22 lags (0..21) using a unimodal weight
lag_weights <- dgamma(0:21, shape = 2, rate = 1.2)
lag_weights <- lag_weights / sum(lag_weights)
cat("Lag weights sum (should be 1):", sum(lag_weights), "\n")
cat("Peak lag:", which.max(lag_weights) - 1, "\n")

# --- DGP function: given a wide temp matrix (n_da x n_days), draw counts ---
simulate_counts <- function(temp_mat, da_long, truth_factors, mmt_da, seed = 42) {
  set.seed(seed)
  n_da   <- nrow(temp_mat)
  n_days <- ncol(temp_mat)
  cat("Simulating", n_da, "DAs ×", n_days, "days × 4 age bands\n")

  # log_RR per DA-day, applying base curve + F1/F2/F3 modifiers
  F1 <- truth_factors$F1; F2 <- truth_factors$F2; F3 <- truth_factors$F3
  # Vectorized: each row is a DA, each col a day
  log_rr_mat <- matrix(0, n_da, n_days)
  for (j in 1:n_days) {
    T_j <- temp_mat[, j]
    base_j <- base_log_rr(T_j, mmt_da)            # length n_da
    heat_ind <- pmax(T_j - mmt_da, 0) > 0
    cold_ind <- pmax(mmt_da - T_j, 0) > 0
    log_rr_mat[, j] <- base_j * (1 +
                                 0.4 * F1 * heat_ind +
                                 0.3 * F2 * cold_ind +
                                 0.2 * F3)
  }

  # Apply 22-day lag convolution (forward-looking smear)
  log_rr_lagged <- matrix(0, n_da, n_days)
  for (i in 1:n_da) {
    log_rr_lagged[i, ] <- stats::filter(log_rr_mat[i, ], lag_weights, sides = 1)
  }
  log_rr_lagged[is.na(log_rr_lagged)] <- 0  # NA at edges -> no lagged effect yet

  # Now apply per (da_idx, date, age_band) → multiply lambda0 by exp(log_rr_lagged)
  # Build a long lookup: da_idx × date_idx × age_band
  date_lookup <- data.table(date = study_dates, date_idx = seq_along(study_dates))
  da_long2 <- merge(da_long, date_lookup, by = "date", all.x = TRUE)
  setkey(da_long2, da_idx, date_idx, age_band)

  # Extract log_rr for each row
  da_long2[, log_rr := log_rr_lagged[cbind(da_idx, date_idx)]]
  da_long2[, lambda := lambda0 * exp(log_rr)]
  da_long2[, n_deaths := rpois(.N, lambda)]

  cat("Total deaths simulated:", sum(da_long2$n_deaths), "\n")
  cat("Mean deaths per DA-day-age:", round(mean(da_long2$n_deaths), 4), "\n")
  cat("% zero days:", round(mean(da_long2$n_deaths == 0) * 100, 1), "%\n")

  return(da_long2)
}

# --- Cross-basis builder (verbatim Gasparrini 2022 supplement) ---
# Temperature: quadratic B-spline, knots at 10/75/90th percentiles of CMA temp
# Lag: natural cubic spline on log scale of 0-21 day window
build_crossbasis <- function(T_series, lag_max = 21) {
  T_knots <- quantile(T_series, probs = c(0.10, 0.75, 0.90), na.rm = TRUE)
  cb <- crossbasis(
    x      = T_series,
    lag    = lag_max,
    argvar = list(fun = "bs", degree = 2, knots = T_knots),
    arglag = list(fun = "ns", knots = logknots(lag_max, nk = 3))
  )
  stopifnot(attr(cb, "argvar")$fun == "bs")  # U4 unit test
  stopifnot(attr(cb, "argvar")$degree == 2)
  cb
}

# --- Strata builders (U5: two variants, qAIC pick) ---
make_strata_A <- function(DA_id, date) {
  # Coarse: DA × year × month — DOW enters as covariate
  factor(paste(DA_id, year(date), month(date), sep = "_"))
}
make_strata_B <- function(DA_id, date) {
  # Fine: DA × year × month × DOW — absorbs day-of-week
  factor(paste(DA_id, year(date), month(date), wday(date), sep = "_"))
}

# --- qAIC for quasi-Poisson gnm fit ---
qaic <- function(fit) {
  phi <- summary(fit)$dispersion
  -2 * as.numeric(logLik(fit)) + 2 * length(coef(fit)) * phi
}

# --- Stage 1 fit for one (CMA, age) pair, returns winner + diagnostics ---
fit_stage1 <- function(cma_age_data, cb_template, cma_label, age_label, verbose = TRUE) {
  if (verbose) cat(sprintf("\n=== Stage 1: %s, age %s ===\n", cma_label, age_label))

  # Both strata variants
  cma_age_data[, strata_A := make_strata_A(DA_id, date)]
  cma_age_data[, strata_B := make_strata_B(DA_id, date)]
  cma_age_data[, dow := factor(wday(date))]
  cma_age_data[, t   := as.integer(date - min(date)) + 1]

  results <- list()
  for (variant in c("A", "B")) {
    strata_col <- if (variant == "A") "strata_A" else "strata_B"
    formula <- if (variant == "A") {
      n_deaths ~ cb + dow + ns(t, df = 20)
    } else {
      n_deaths ~ cb + ns(t, df = 20)
    }
    if (verbose) cat(sprintf("  Variant %s: %d strata, %d rows\n",
                              variant,
                              length(unique(cma_age_data[[strata_col]])),
                              nrow(cma_age_data)))

    cma_age_data[, strata_use := get(strata_col)]

    fit <- try(gnm(formula,
                   data      = cma_age_data,
                   family    = quasipoisson(),
                   eliminate = strata_use),
               silent = TRUE)

    if (inherits(fit, "try-error")) {
      if (verbose) cat(sprintf("  Variant %s: FAILED to converge\n", variant))
      results[[variant]] <- NULL
      next
    }

    results[[variant]] <- list(
      fit       = fit,
      qaic      = qaic(fit),
      n_strata  = length(unique(cma_age_data[[strata_col]])),
      mean_deaths_per_stratum = sum(cma_age_data$n_deaths) /
                                length(unique(cma_age_data[[strata_col]]))
    )
    if (verbose) cat(sprintf("  Variant %s: qAIC = %.1f\n", variant, results[[variant]]$qaic))
  }

  # Pick winner
  qaic_A <- if (!is.null(results$A)) results$A$qaic else Inf
  qaic_B <- if (!is.null(results$B)) results$B$qaic else Inf
  winner_var <- if (qaic_A < qaic_B) "A" else "B"
  if (verbose) cat(sprintf("  Winner: variant %s\n", winner_var))

  cb_idx <- grep("^cb", names(coef(results[[winner_var]]$fit)))
  list(
    cma             = cma_label,
    age             = age_label,
    winner_variant  = winner_var,
    coef            = coef(results[[winner_var]]$fit)[cb_idx],
    vcov            = vcov(results[[winner_var]]$fit)[cb_idx, cb_idx],
    n_strata_A      = if (!is.null(results$A)) results$A$n_strata else NA,
    n_strata_B      = if (!is.null(results$B)) results$B$n_strata else NA,
    qaic_A          = qaic_A,
    qaic_B          = qaic_B,
    mean_dps_winner = results[[winner_var]]$mean_deaths_per_stratum,
    cb_template     = cb_template
  )
}

# --- Reduce: collapse 16-dim cross-basis to 4-dim cumulative cross-section ---
# cen = CMA-level reference temp at this stage; DA-specific re-centering happens in Stage 4 (U11)
reduce_fit <- function(stage1_result, ref_temp, verbose = TRUE) {
  cb_t <- stage1_result$cb_template

  # Rebuild a fake gnm-like object that crossreduce accepts: it needs coef + vcov at cb basis indices
  # crossreduce() signature: crossreduce(basis, model = NULL, coef = NULL, vcov = NULL, ...)
  red <- crossreduce(
    basis = cb_t,
    coef  = stage1_result$coef,
    vcov  = stage1_result$vcov,
    type  = "overall",
    cen   = ref_temp
  )
  if (verbose) cat(sprintf("  reduced %s/%s: 16-dim -> 4-dim, cen=%.1f C\n",
                            stage1_result$cma, stage1_result$age, ref_temp))
  list(
    cma         = stage1_result$cma,
    age         = stage1_result$age,
    theta_star  = coef(red),       # length 4
    V_star      = vcov(red),       # 4x4
    cen         = ref_temp,
    reduced_obj = red
  )
}

# --- Stage 2 multivariate meta-regression with mixmeta ---
# Random intercept per CMA (U7), REML, no interactions (U8)
fit_stage2 <- function(reduced_list, cma_predictors_df, verbose = TRUE) {
  # reduced_list = list of reduce_fit() outputs, one per (cma, age)
  # cma_predictors_df = data.frame with CMA-level PC1, PC2, PC3 (population-weighted)

  # Build the response matrix (n_obs x 4) and S list
  n_obs <- length(reduced_list)
  theta_mat <- t(sapply(reduced_list, function(x) x$theta_star))
  V_list    <- lapply(reduced_list, function(x) x$V_star)

  # Build predictor data.frame
  pred_df <- data.table(
    CMA       = sapply(reduced_list, function(x) x$cma),
    age_band  = sapply(reduced_list, function(x) x$age),
    obs_idx   = seq_len(n_obs)
  )
  pred_df <- merge(pred_df, cma_predictors_df, by = "CMA", all.x = TRUE)
  setorder(pred_df, obs_idx)

  if (verbose) {
    cat(sprintf("  Stage 2: %d obs, %d CMAs, %d age bands\n",
                n_obs, length(unique(pred_df$CMA)), length(unique(pred_df$age_band))))
    cat(sprintf("  Ψ unique params: %d (4-dim symmetric)\n", 4*5/2))
    cat(sprintf("  Obs/param ratio: %.2f (target >= 2)\n", n_obs / (4*5/2)))
  }

  # mixmeta call — U7 multilevel
  # NOTE: with N_CMA = 1 (Toronto-only pilot), random = ~1|CMA is degenerate.
  # In that case fall back to fixed-effects only.
  n_cma <- length(unique(pred_df$CMA))
  if (n_cma >= 2) {
    fit <- mixmeta(
      formula = theta_mat ~ age_band + PC1 + PC2 + PC3,
      random  = ~ 1 | CMA,
      S       = V_list,
      data    = pred_df,
      method  = "reml"
    )
  } else {
    if (verbose) cat("  N_CMA=1: random intercept degenerate. Falling back to fixed effects.\n")
    fit <- mixmeta(
      formula = theta_mat ~ age_band + PC1 + PC2 + PC3,
      S       = V_list,
      data    = pred_df,
      method  = "fixed"
    )
  }

  # U9 convergence diagnostics
  if (verbose) {
    cat(sprintf("  Converged: %s\n", isTRUE(fit$converged)))
    psi_chol <- tryCatch(chol(fit$Psi %||% matrix(0,4,4)), error = function(e) NULL)
    cat(sprintf("  Ψ Cholesky pass: %s\n", !is.null(psi_chol)))
  }

  list(fit = fit, pred_df = pred_df, theta_mat = theta_mat, V_list = V_list)
}

# --- DA-level PCA: 17 vulnerability vars -> 3 components (sanity check U15 inside) ---
fit_da_pca <- function(Z_matrix, da_ids, verbose = TRUE) {
  # Z_matrix: n_da × 17 (the 17 vulnerability vars)
  # da_ids: length n_da DAUIDs

  pca <- prcomp(Z_matrix, scale. = TRUE)
  var_explained <- summary(pca)$importance["Proportion of Variance", 1:3]
  cum_var       <- summary(pca)$importance["Cumulative Proportion", 3]

  da_scores <- data.table(
    DAUID = da_ids,
    PC1   = pca$x[, 1],
    PC2   = pca$x[, 2],
    PC3   = pca$x[, 3]
  )

  if (verbose) {
    cat(sprintf("  PCA: var explained PC1-3 = %.1f%%, %.1f%%, %.1f%% (cum %.1f%%)\n",
                100*var_explained[1], 100*var_explained[2], 100*var_explained[3],
                100*cum_var))
    stopifnot(cum_var > 0.5)  # if first 3 PCs explain <50%, something is off
  }

  list(pca = pca, scores = da_scores, var_explained = var_explained)
}

# --- DA-level predict from Stage 2 ---
# Plug DA-level PC scores into the fitted meta-regression -> DA-specific θ* (4-vec)
predict_da_theta <- function(stage2_obj, da_scores, age_bands = c("age_0_64","age_65_74","age_75_84","age_85p")) {
  # Returns: data.table with DAUID × age_band × (theta1..theta4)
  out_list <- list()
  for (a in age_bands) {
    newdat <- data.table(age_band = a,
                          PC1 = da_scores$PC1,
                          PC2 = da_scores$PC2,
                          PC3 = da_scores$PC3)
    pred <- predict(stage2_obj$fit, newdata = newdat)
    # pred is n_da × 4
    out_list[[a]] <- data.table(DAUID = da_scores$DAUID,
                                 age_band = a,
                                 theta1 = pred[, 1], theta2 = pred[, 2],
                                 theta3 = pred[, 3], theta4 = pred[, 4])
  }
  rbindlist(out_list)
}

# --- Per-DA two-pass MMT (U11) ---
# Pass 1: predict at CMA-median, find DA-specific minimum mortality temperature
# Pass 2: re-center at DA-MMT, compute final RR and AF
compute_da_mmt <- function(da_theta_row, cb_template, temp_range_da, cma_median) {
  # da_theta_row: length-4 numeric
  # temp_range_da: temp values at which to evaluate (e.g., DA's own temp quantiles)
  # cma_median: CMA median temp (used for first pass cen)

  # First pass: find MMT
  pred1 <- crosspred(basis = cb_template,
                     coef  = da_theta_row,
                     vcov  = diag(1e-8, 4),  # dummy small vcov (MC happens separately)
                     at    = temp_range_da,
                     cen   = cma_median)
  mmt_da <- pred1$predvar[which.min(pred1$allfit)]
  return(mmt_da)
}

# --- Monte Carlo at fixed-effect coefficient level (U12) ---
# Sample 1000 draws of Stage 2 coefs, propagate per-DA, take quantiles aggregate
monte_carlo_ci <- function(stage2_obj, da_scores, n_sim = 1000, verbose = TRUE) {
  if (verbose) cat(sprintf("  Monte Carlo: %d draws of Stage 2 coefs\n", n_sim))

  beta_hat <- as.vector(coef(stage2_obj$fit))
  V_hat    <- vcov(stage2_obj$fit)

  beta_sims <- MASS::mvrnorm(n_sim, mu = beta_hat, Sigma = V_hat)

  if (verbose) cat(sprintf("  beta_sims: %d × %d\n", nrow(beta_sims), ncol(beta_sims)))
  beta_sims
}

# === MASTER PIPELINE — Toronto pilot, full run ===
# Calls every scaffold function in order. Requires da_daily_temp to exist.

run_pilot <- function(verbose = TRUE) {
  t0 <- Sys.time()

  # ============================================================
  # STEP A — Load + reshape temperature data into wide matrix
  # ============================================================
  if (verbose) cat("\n[A] Reshaping temperature data...\n")
  stopifnot(exists("da_daily_temp"), nrow(da_daily_temp) > 0)

  # da_daily_temp comes from Earth Engine: DAUID × date × tmax × tmin
  da_daily_temp[, date := as.Date(date)]
  da_daily_temp[, tmean_C := (tmax + tmin) / 2]

  # Wide matrix: n_da rows × n_days cols
  temp_wide <- dcast(da_daily_temp, DAUID ~ date, value.var = "tmean_C")
  setnames(temp_wide, "DAUID", "DAUID_str")

  # Align rows to truth_factors / da_age order
  temp_wide <- temp_wide[match(truth_factors$DAUID, DAUID_str)]
  stopifnot(all(temp_wide$DAUID_str == truth_factors$DAUID))

  temp_mat <- as.matrix(temp_wide[, -"DAUID_str"])
  cat(sprintf("  temp_mat: %d × %d\n", nrow(temp_mat), ncol(temp_mat)))
  stopifnot(nrow(temp_mat) == n_da, ncol(temp_mat) == length(study_dates))

  # ============================================================
  # STEP B — DA-specific MMT priors (80th pct of each DA's temp distribution)
  # ============================================================
  if (verbose) cat("\n[B] Computing DA-level MMT anchors...\n")
  mmt_da <- apply(temp_mat, 1, quantile, probs = 0.80, na.rm = TRUE)
  cat(sprintf("  MMT_da range: %.1f to %.1f °C\n", min(mmt_da), max(mmt_da)))

  # ============================================================
  # STEP C — Simulate mortality counts
  # ============================================================
  if (verbose) cat("\n[C] Simulating mortality counts...\n")
  sim_data <- simulate_counts(temp_mat, da_long, truth_factors, mmt_da)

  # Attach DAUID + temp to sim_data for Stage 1
  date_lookup <- data.table(date = study_dates, date_idx = seq_along(study_dates))
  truth_lookup <- truth_factors[, .(da_idx, DAUID)]
  sim_data <- merge(sim_data, truth_lookup, by = "da_idx")

  # Attach temperature per (da_idx, date)
  temp_long <- melt(as.data.table(cbind(da_idx = 1:n_da, as.data.table(temp_mat))),
                    id.vars = "da_idx", variable.name = "date_chr", value.name = "tmean_C")
  temp_long[, date := as.Date(date_chr)]
  temp_long[, date_chr := NULL]
  sim_data <- merge(sim_data, temp_long, by = c("da_idx", "date"))

  setnames(sim_data, "DAUID", "DA_id")
  cat(sprintf("  sim_data merged: %d rows\n", nrow(sim_data)))

  # ============================================================
  # STEP D — Stage 1: fit per (CMA, age) with both strata variants
  # ============================================================
  if (verbose) cat("\n[D] Stage 1 fits (4 age bands × 2 variants = 8 candidate fits)...\n")

  cma_temp_series <- sim_data[, .(tmean_C = mean(tmean_C, na.rm = TRUE)), by = date]$tmean_C
  cb_template <- build_crossbasis(cma_temp_series, lag_max = 21)
  cat(sprintf("  cross-basis built: %d cols\n", ncol(cb_template)))

  age_bands <- c("age_0_64", "age_65_74", "age_75_84", "age_85p")
  stage1_results <- list()
  for (a in age_bands) {
    sub <- sim_data[age_band == a]
    setorder(sub, DA_id, date)
    cb_sub <- build_crossbasis(sub$tmean_C, lag_max = 21)
    fit <- fit_stage1(sub, cb_sub, cma_label = "Toronto", age_label = a, verbose = TRUE)
    stage1_results[[a]] <- fit
  }

  # ============================================================
  # STEP E — Reduce 16-dim -> 4-dim, fit Stage 2
  # ============================================================
  if (verbose) cat("\n[E] Reduce + Stage 2 meta-regression...\n")
  cma_median <- median(cma_temp_series, na.rm = TRUE)
  reduced_list <- lapply(stage1_results, reduce_fit, ref_temp = cma_median, verbose = TRUE)

  # DA-level PCA
  pca_result <- fit_da_pca(Z, truth_factors$DAUID, verbose = TRUE)

  # CMA-level PC scores (population-weighted average across DAs)
  pop_total <- da_age$total
  cma_pcs <- data.table(
    CMA = "Toronto",
    PC1 = weighted.mean(pca_result$scores$PC1, pop_total),
    PC2 = weighted.mean(pca_result$scores$PC2, pop_total),
    PC3 = weighted.mean(pca_result$scores$PC3, pop_total)
  )
  cat(sprintf("  Toronto CMA PCs: PC1=%.3f PC2=%.3f PC3=%.3f\n",
              cma_pcs$PC1, cma_pcs$PC2, cma_pcs$PC3))

  stage2 <- fit_stage2(reduced_list, cma_pcs, verbose = TRUE)

  # ============================================================
  # STEP F — Downscale to DA
  # ============================================================
  if (verbose) cat("\n[F] Downscaling to DA level...\n")
  da_theta <- predict_da_theta(stage2, pca_result$scores, age_bands)
  cat(sprintf("  DA-level θ*: %d rows (DAUID × age)\n", nrow(da_theta)))

  # ============================================================
  # STEP G — Monte Carlo confidence intervals
  # ============================================================
  if (verbose) cat("\n[G] Monte Carlo CIs (1000 draws)...\n")
  beta_sims <- monte_carlo_ci(stage2, pca_result$scores, n_sim = 1000, verbose = TRUE)

  # ============================================================
  # RETURN PILOT BUNDLE
  # ============================================================
  elapsed <- difftime(Sys.time(), t0, units = "mins")
  cat(sprintf("\n=== Pilot run complete: %.1f min ===\n", as.numeric(elapsed)))

  list(
    temp_mat        = temp_mat,
    mmt_da          = mmt_da,
    sim_data        = sim_data,
    stage1_results  = stage1_results,
    reduced_list    = reduced_list,
    pca_result      = pca_result,
    cma_pcs         = cma_pcs,
    stage2          = stage2,
    da_theta        = da_theta,
    beta_sims       = beta_sims,
    runtime_min     = as.numeric(elapsed)
  )
}

qaic <- function(fit) {
  phi <- summary(fit)$dispersion
  deviance(fit)/phi + 2 * length(coef(fit))
}

"""function needs metadata that's normally autodetected

Add model.link="log"
"""

reduce_fit <- function(stage1_result, ref_temp, verbose = TRUE) {
  cb_t <- stage1_result$cb_template
  red  <- crossreduce(
    basis = cb_t,
    coef  = stage1_result$coef,
    vcov  = stage1_result$vcov,
    model.link = "log",        # <- the missing piece
    type  = "overall",
    cen   = ref_temp
  )
  list(cma = stage1_result$cma, age = stage1_result$age,
       theta_star = coef(red), V_star = vcov(red),
       cen = ref_temp, reduced_obj = red)
}
